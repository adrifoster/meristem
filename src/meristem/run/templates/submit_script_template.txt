#!/bin/bash
#PBS -N {{ job_name }}
#PBS -q {{ queue }}
#PBS -l select={{ select }}:ncpus={{ ncpus }}:mem={{ memory }}
#PBS -l walltime={{ walltime }}
#PBS -A {{ project }}
#PBS -o {{ log_file }}
#PBS -j oe
#PBS -k eod
#PBS -M {{ user }}
{% if depend_job_id %}
#PBS -W depend=afterok:{{ depend_job_id }}
{% endif %}
{{ extra }}

# activate conda environment
source $(conda info --base)/etc/profile.d/conda.sh
conda activate meristem

set -euo pipefail
 
RUN_ROOT={{ root | shquote }}
STAGE={{ name | shquote }}
trap 'meristem run --root "$RUN_ROOT" --fail --stage "$STAGE"' ERR
 
echo "=== Run: {{ run_id }} / stage: {{ name }} ==="
echo "=== Started: $(date) ==="

{% if not skip_script %}
# create, configure, and build the CIME case
# strict mode is enforced even if the user's script doesn't set it
bash -c 'set -euo pipefail; source "$0" "$@"' {{ script | shquote }} {{ case_root | shquote }}{% if previous_case %} {{ previous_case | shquote }} {{ previous_case_root | shquote }}{% endif +%}
{% endif %}

{% if cime %}
# submit the CIME case; capture the final job ID (short-term archiver)
cd {{ case_root | shquote }}
./case.submit --resubmit-immediate{% if submit_args %} -a {{ submit_args | shquote }}{% endif %}
JOB_IDS=$(./xmlquery JOB_IDS)
FINAL_JOB_ID=${JOB_IDS##*:}
echo "CIME submitted, final job ID: $FINAL_JOB_ID"
 
meristem run --root "$RUN_ROOT" --submit-advance --stage "$STAGE" --cime-job-id $FINAL_JOB_ID
meristem run --root "$RUN_ROOT" --submit-fail --stage "$STAGE" --cime-job-id $FINAL_JOB_ID
{% else %}
meristem run --root "$RUN_ROOT" --advance --stage "$STAGE"
{% endif %}

echo "=== Setup complete: $(date) ==="
