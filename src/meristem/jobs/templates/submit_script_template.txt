#PBS -N {{ job_name }}

#PBS -l walltime={{ walltime }}
#PBS -A {{ project }}
#PBS -o {{ log_file }}
#PBS -j oe
#PBS -k eod
#PBS -M {{ user }}
{% if depend_job_id %}
#PBS -W depend=afterok:{{ depend_job_id }}
{% endif %}
{{ extra }}

# activate conda environment
source $(conda info --base)/etc/profile.d/conda.sh
conda activate tethering

set -euo pipefail
 
trap 'clm-run --root {{ root }} --fail --stage {{ name }}' ERR
 
echo "=== Run: {{ run_id }} / stage: {{ name }} ==="
echo "=== Started: $(date) ==="

{% if not skip_script %}
# create, configure, and build the CIME case
# strict mode is enforced even if the user's script doesn't set it
bash -c 'set -euo pipefail; source "$0" "$@"' {{ script }} {{ case_root }} {{ previous_case }}
{% endif %}

{% if cime %}
# submit the CIME case; capture the final job ID (short-term archiver)
cd {{ case_root }}
./case.submit --resubmit-immediate
JOB_IDS=$(./xmlquery JOB_IDS)
FINAL_JOB_ID=${JOB_IDS##*:}
echo "CIME submitted, final job ID: $FINAL_JOB_ID"
 
clm-run --root {{ root }} --submit-advance --stage {{ name }} --cime-job-id $FINAL_JOB_ID
clm-run --root {{ root }} --submit-fail --stage {{ name }} --cime-job-id $FINAL_JOB_ID
{% else %}
clm-run --root {{ root }} --advance --stage {{ name }}
{% endif %}

echo "=== Setup complete: $(date) ==="
